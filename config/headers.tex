%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% Headers and Footers Configuration
% fancyhdr settings for page headers/footers
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\pagestyle{fancy}
\fancyhf{}

% Headers (chapter title on odd pages only)
\fancyhead[LE]{}
\fancyhead[RO]{\rightmark}

% Footers (page number centered)
\fancyfoot[C]{\thepage}

% Header/footer rules
\renewcommand{\headrulewidth}{0.5pt}
\renewcommand{\footrulewidth}{0pt}

% Custom chapter mark (remove "Chapter" word from header)
\makeatletter
\renewcommand{\chaptermark}[1]{\markboth{\thechapter.\ #1}{}}
\makeatother
